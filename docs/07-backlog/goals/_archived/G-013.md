## Intent

**Why:** Cockpit's `ThemeToggle` uses `motion/react` directly and is 100% generic — every Pcampus service needs a theme toggle. Moving it to pcampus-ui gives all services the same animated toggle without rebuilding.

**Done when:**
- `import { ThemeToggle } from '@pcampus-studio/ui'` works
- Animation uses `var(--pcampus-duration-fast)` and `var(--pcampus-ease-standard)` — no hardcoded timing
- Cockpit can delete `components/theme-toggle.tsx` and replace with library import

**Unblocks:** Cockpit G-032 (delete local ThemeToggle)

## Architecture

`motion` is a **peerDependency** — consumers install it themselves. Cockpit already has `"motion": "^12.42.0"`.

```json
// package.json
"peerDependencies": {
  "react": ">=18",
  "motion": ">=11"
}
```

## Component spec

```tsx
// src/primitives/theme-toggle.tsx
'use client';

import { useEffect, useState } from 'react';
import { motion } from 'motion/react';

export function ThemeToggle() {
  const [mounted, setMounted] = useState(false);
  const [theme, setTheme] = useState<'light' | 'dark'>('light');

  useEffect(() => {
    setMounted(true);
    const isDark = document.documentElement.classList.contains('dark');
    setTheme(isDark ? 'dark' : 'light');
  }, []);

  const toggleTheme = () => {
    const next = theme === 'light' ? 'dark' : 'light';
    setTheme(next);
    document.documentElement.classList.toggle('dark', next === 'dark');
    localStorage.setItem('theme', next);
  };

  if (!mounted) {
    return <div className="h-9 w-9 rounded-full border border-pcampus-hairline bg-pcampus-surface-card/50" />;
  }

  const iconTransition = { duration: 0.15, ease: [0.16, 1, 0.3, 1] }; // pcampus-duration-fast + pcampus-ease-standard

  return (
    <button
      onClick={toggleTheme}
      className="relative flex h-9 w-9 items-center justify-center rounded-full border border-pcampus-hairline bg-pcampus-surface-card/60 text-pcampus-ink shadow-sm transition-all hover:bg-pcampus-canvas-soft hover:shadow backdrop-blur-sm cursor-pointer"
      aria-label={`Switch to ${theme === 'light' ? 'dark' : 'light'} mode`}
    >
      {/* Sun icon */}
      <motion.div
        initial={false}
        animate={{ scale: theme === 'light' ? 1 : 0, rotate: theme === 'light' ? 0 : 90, opacity: theme === 'light' ? 1 : 0 }}
        transition={iconTransition}
        className="absolute"
      >
        <SunIcon />
      </motion.div>
      {/* Moon icon */}
      <motion.div
        initial={false}
        animate={{ scale: theme === 'dark' ? 1 : 0, rotate: theme === 'dark' ? 0 : -90, opacity: theme === 'dark' ? 1 : 0 }}
        transition={iconTransition}
        className="absolute"
      >
        <MoonIcon />
      </motion.div>
    </button>
  );
}

function SunIcon() {
  return (
    <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={1.5} stroke="currentColor" className="h-[18px] w-[18px]">
      <path strokeLinecap="round" strokeLinejoin="round" d="M12 3v2.25m6.364.386-1.591 1.591M21 12h-2.25m-.386 6.364-1.591-1.591M12 18.75V21m-4.773-4.227-1.591 1.591M5.25 12H3m4.227-4.773L5.636 5.636M15.75 12a3.75 3.75 0 1 1-7.5 0 3.75 3.75 0 0 1 7.5 0Z" />
    </svg>
  );
}

function MoonIcon() {
  return (
    <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={1.5} stroke="currentColor" className="h-[18px] w-[18px]">
      <path strokeLinecap="round" strokeLinejoin="round" d="M21.752 15.002A9.72 9.72 0 0 1 18 15.75c-5.385 0-9.75-4.365-9.75-9.75 0-1.33.266-2.597.748-3.752A9.753 9.753 0 0 0 3 11.25C3 16.635 7.365 21 12.75 21a9.753 9.753 0 0 0 9.002-5.998Z" />
    </svg>
  );
}
```
