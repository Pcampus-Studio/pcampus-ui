## Component specs

### Card

```tsx
export interface CardProps extends React.HTMLAttributes<HTMLDivElement> {
  padding?: 'none' | 'sm' | 'md' | 'lg';  // default: 'md'
}

export function Card({ padding = 'md', className, ...props }: CardProps) {
  return (
    <div
      className={clsx(
        'rounded-pcampus-lg border border-pcampus-hairline bg-pcampus-surface-card',
        padding === 'sm' && 'p-4',
        padding === 'md' && 'p-6',
        padding === 'lg' && 'p-8',
        className,
      )}
      {...props}
    />
  );
}

export function CardHeader({ className, ...props }) { ... }    // flex col gap-1.5 p-6
export function CardTitle({ className, ...props }) { ... }     // text-lg font-semibold text-pcampus-ink
export function CardDescription({ className, ...props }) { ... } // text-sm text-pcampus-muted
export function CardContent({ className, ...props }) { ... }   // p-6 pt-0
export function CardFooter({ className, ...props }) { ... }    // flex items-center p-6 pt-0
```
